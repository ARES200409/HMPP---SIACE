@rrhh_bp.route('/reporte/anual//') @login_required @role_required('RRHH', 'AdministradorLegajos', 'Sistemas') def generar_reporte_anual(id_personal, anio): from weasyprint import HTML import io import re # 🔥 IMPORTAMOS RE PARA EXTRAER EL TEXTO ENTRE PARÉNTESIS from app.database import get_db_read from datetime import datetime repo_personal = SqlServerPersonalRepository() repo_planilla = PlanillaRepository() conn = get_db_read() cursor = conn.cursor() # Función auxiliar para extraer lo que está entre paréntesis (Ej: BONESP, REUNIF) def obtener_abreviatura(nombre_completo): match = re.search(r'\((.*?)\)', nombre_completo) return match.group(1) if match else nombre_completo try: # 1. OBTENER DATOS PERSONALES Y EL DNI PARA EL BARRIDO GLOBAL cursor.execute(""" SELECT TOP 1 dp.nombre_completo, dp.cargo_actual, dp.sistema_pensionario, dp.dni_trabajador, dp.nivel, dp.condicion_laboral, p.fecha_ingreso, dp.observaciones, p.id_tipo_contrato FROM detalle_planilla dp INNER JOIN planillas pl ON dp.id_planilla = pl.id_planilla INNER JOIN personal p ON dp.id_personal = p.id_personal WHERE dp.id_personal = ? AND pl.anio = ? ORDER BY pl.mes DESC """, (id_personal, anio)) row_emp = cursor.fetchone() if not row_emp: flash('No hay datos en planilla para este trabajador en el año seleccionado.', 'danger') return redirect(url_for('rrhh.listar_personal')) dni_target = row_emp[3] # 🔥 ESTE DNI SE USARÁ PARA JUNTAR TODAS LAS PLANILLAS DEL AÑO personal_dict = { 'apellidos': row_emp[0], 'nombres': '', 'cargo_actual': row_emp[1], 'sistema_pensionario': row_emp[2] or 'ONP', 'dni': dni_target, 'nivel': row_emp[4] if row_emp[4] else '-', 'regimen_laboral': str(row_emp[8]) if row_emp[8] is not None else '', 'regimen_laboral_texto': row_emp[5] or 'No definido', 'fecha_ingreso': row_emp[6].strftime('%d/%m/%Y') if row_emp[6] else '-', } cargos_anio = repo_planilla.obtener_cargos_anio(id_personal, anio) # 2. OBTENER NOMBRES ÚNICOS PARA LAS COLUMNAS DINÁMICAS (Usando DNI y Abreviaturas) cursor.execute(""" SELECT DISTINCT c.nombre_concepto, c.tipo FROM detalle_conceptos_trabajador dc INNER JOIN catalogo_conceptos c ON dc.id_concepto = c.id_concepto INNER JOIN detalle_planilla dp ON dc.id_detalle = dp.id_detalle INNER JOIN planillas pl ON dp.id_planilla = pl.id_planilla WHERE dp.dni_trabajador = ? AND pl.anio = ? """, (dni_target, anio)) # 🔥 Búsqueda ampliada por DNI conceptos = cursor.fetchall() bonos_unicos = [] descuentos_unicos = [] for c_nom, c_tipo in conceptos: abrev = obtener_abreviatura(c_nom) if c_tipo == 'INGRESO' and abrev not in bonos_unicos: bonos_unicos.append(abrev) elif c_tipo == 'DESCUENTO' and abrev not in descuentos_unicos: descuentos_unicos.append(abrev) # Valores por defecto en caso no tenga conceptos extra en todo el año if not bonos_unicos: bonos_unicos = ['BONESP', 'BONODIF'] # 3. CONSTRUIR LA TABLA MES A MES RECOPILANDO TODO pagos_anuales = [] for i in range(1, 13): # 🔥 BUSCAMOS TODAS LAS PLANILLAS DE ESE MES POR DNI (CAS + Nombrado, etc.) sql_mes = """ SELECT dp.dias_computables, dp.dias_falta, dp.sueldo_basico, dp.reintegros, dp.asignacion_familiar, dp.total_ingresos, dp.monto_pension, dp.renta_4ta_5ta, dp.judiciales, dp.faltas_tardanzas, dp.total_descuentos, dp.neto_pagar, dp.aporte_essalud, dp.id_detalle, dp.observaciones, dp.d_viatico, dp.desc_sindicato, dp.essalud_vida, dp.sctr, dp.sctr_onp, dp.cts, dp.rps, dp.bonificaciones, dp.otros_descuentos FROM detalle_planilla dp INNER JOIN planillas pl ON dp.id_planilla = pl.id_planilla WHERE dp.dni_trabajador = ? AND pl.anio = ? AND pl.mes = ? """ cursor.execute(sql_mes, (dni_target, anio, i)) filas_mes = cursor.fetchall() # 🔥 FETCHALL() obtiene TODAS las planillas de ese mes # Diccionario en 0 por defecto para evitar Crash de 'basica' pago = { 'mes': i, 'vacio': True, 'd_lab': 0, 'd_fal': 0, 'basica': 0.0, 'reintegro': 0.0, 'bonif': 0.0, 'asig_fam': 0.0, 'otros_ing': 0.0, 'ing_bruto': 0.0, 's_pens': 0.0, 'renta': 0.0, 'judic': 0.0, 'tardanza': 0.0, 'otros_dscto': 0.0, 'tot_dscto': 0.0, 'neto': 0.0, 'rps': 0.0, 'essalud_9': 0.0, 'sctr_salud': 0.0, 'sctr_pension': 0.0, 'cts': 0.0, 'dict_bonos': {}, 'dict_descuentos': {}, 'medic': 0, 'vacac': 0, 'permis': 0, 'susp': 0, 'escolar': 0.0, 'f_patrias': 0.0, 'navidad': 0.0, 'observaciones': '' } if filas_mes: pago['vacio'] = False obs_list = [] # 🔥 JUNTAMOS (SUMAMOS) LOS DATOS DE TODAS LAS PLANILLAS DEL MES for r in filas_mes: pago['d_lab'] += float(r[0] or 0) pago['d_fal'] += float(r[1] or 0) pago['basica'] += float(r[2] or 0) pago['reintegro'] += float(r[3] or 0) pago['asig_fam'] += float(r[4] or 0) pago['ing_bruto'] += float(r[5] or 0) pago['s_pens'] += float(r[6] or 0) pago['renta'] += float(r[7] or 0) pago['judic'] += float(r[8] or 0) pago['tardanza'] += float(r[9] or 0) pago['tot_dscto'] += float(r[10] or 0) pago['neto'] += float(r[11] or 0) pago['essalud_9'] += float(r[12] or 0) id_det = r[13] if r[14]: obs_list.append(r[14]) viatico = float(r[15] or 0) sindicato = float(r[16] or 0) vida = float(r[17] or 0) pago['otros_ing'] += viatico pago['sctr_salud'] += float(r[18] or 0) pago['sctr_pension'] += float(r[19] or 0) pago['cts'] += float(r[20] or 0) pago['rps'] += float(r[21] or 0) pago['bonif'] += float(r[22] or 0) pago['otros_dscto'] += float(r[23] or 0) + sindicato + vida # Extraemos y sumamos conceptos de esta planilla específica cursor.execute(""" SELECT c.nombre_concepto, c.tipo, dc.monto FROM detalle_conceptos_trabajador dc INNER JOIN catalogo_conceptos c ON dc.id_concepto = c.id_concepto WHERE dc.id_detalle = ? """, (id_det,)) for c_nom, c_tipo, c_monto in cursor.fetchall(): monto_flt = float(c_monto or 0) abrev = obtener_abreviatura(c_nom) if c_tipo == 'INGRESO': pago['dict_bonos'][abrev] = pago['dict_bonos'].get(abrev, 0.0) + monto_flt else: pago['dict_descuentos'][abrev] = pago['dict_descuentos'].get(abrev, 0.0) + monto_flt # Inyectamos Sindicato y Vida como columnas dinámicas if sindicato > 0: pago['dict_descuentos']['Cuota Sindical'] = pago['dict_descuentos'].get('Cuota Sindical', 0.0) + sindicato if vida > 0: pago['dict_descuentos']['Essalud Vida'] = pago['dict_descuentos'].get('Essalud Vida', 0.0) + vida pago['observaciones'] = " | ".join(filter(None, obs_list)) pagos_anuales.append(pago) # Agregamos las cabeceras fijas a la lista si se utilizaron en algún momento del año if any(float(p['dict_descuentos'].get('Cuota Sindical', 0)) > 0 for p in pagos_anuales) and 'Cuota Sindical' not in descuentos_unicos: descuentos_unicos.append('Cuota Sindical') if any(float(p['dict_descuentos'].get('Essalud Vida', 0)) > 0 for p in pagos_anuales) and 'Essalud Vida' not in descuentos_unicos: descuentos_unicos.append('Essalud Vida') # 4. RENDERIZAR EL HTML html_content = render_template( 'record_laboral/pdf_record_anual.html', personal=personal_dict, cargos_anio=cargos_anio, pagos=pagos_anuales, anio=anio, bonos_unicos=bonos_unicos, descuentos_unicos=descuentos_unicos, fecha_hoy=datetime.now() ) pdf_bytes = HTML(string=html_content, base_url=request.base_url).write_pdf() nombre_archivo = f"RECORD_ANUAL_{anio}_{personal_dict['dni']}.pdf" datos_record = { 'id_personal': id_personal, 'nombre_archivo': nombre_archivo, 'descripcion': f"Récord Laboral Anual - Consolidado {anio}", 'fecha_inicio': f"{anio}-01-01", 'fecha_fin_vencimiento': f"{anio}-12-31" } repo_personal.add_record_laboral(datos_record, pdf_bytes) return send_file(io.BytesIO(pdf_bytes), mimetype='application/pdf', as_attachment=False, download_name=nombre_archivo) except Exception as e: print(f"🔥 Error Crítico en Récord: {str(e)}") flash(f'Error técnico: {str(e)}', 'danger') return redirect(url_for('rrhh.listar_personal')) finally: conn.close()
