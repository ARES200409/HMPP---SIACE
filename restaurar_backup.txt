--  RESTAURACIÓN DE PRUEBA EN NUEVA BASE DE DATOS
USE [master];
GO

RESTORE DATABASE [BaseDatos_Test] -- <--- Nombre de la nueva BD de prueba
FROM DISK = N'C:\Backups_Escalafon\Legajo_20260128_101327.bak'  <--- Nombre del backup realizado (Legajo_20260128_101327.bak) o si es otro
WITH 
    -- 1. Movemos el archivo de DATOS a una nueva ubicación física
    MOVE N'BaseDatosDiresa' 
    TO N'C:\Program Files\Microsoft SQL Server\MSSQL17.SQLEXPRESS\MSSQL\DATA\BaseDatos_Test.mdf',

    -- 2. Movemos el archivo de LOG a una nueva ubicación física
    MOVE N'BaseDatosDiresa_log' 
    TO N'C:\Program Files\Microsoft SQL Server\MSSQL17.SQLEXPRESS\MSSQL\DATA\BaseDatos_Test_log.ldf',

    REPLACE, -- Permite crear la BD si ya existiera una de prueba anterior
    STATS = 10; -- Nos muestra el progreso cada 10%
GO


USE [master];
GO
--  Verificando que los datos de prueba sean iguales a los reales
SELECT 'REAL' AS Origen, COUNT(*) AS TotalRegistros FROM BaseDatosDiresa.dbo.bitacora
UNION ALL
SELECT 'TEST' AS Origen, COUNT(*) AS TotalRegistros FROM BaseDatos_Test.dbo.bitacora;
GO

--  Buscando al "culpable" de la diferencia pero es normal por 
--  el Backup ultimo que se realizo y se guardo
SELECT TOP 1 * FROM BaseDatosDiresa.dbo.bitacora 
ORDER BY id_bitacora DESC;
GO



--  ELIMINACIÓN SEGURA DE LA BASE DE DATOS DE PRUEBA
USE [master];
GO

-- 1. Forzamos el cierre de cualquier conexión activa a la BD de prueba
-- (Esto evita el error de "La base de datos está en uso")
ALTER DATABASE [BaseDatos_Test] 
SET SINGLE_USER 
WITH ROLLBACK IMMEDIATE;
GO

-- 2. Eliminamos la base de datos y sus archivos físicos (.mdf y .ldf)
DROP DATABASE [BaseDatos_Test];
GO

PRINT '---[ÉXITO]: BaseDatos_Test ha sido eliminada y el espacio en disco liberado.';
